        function calculateMacroGoals() {

            state.profile.gender = document.getElementById('gender').value;

            state.profile.age = parseFloat(document.getElementById('age').value) || 25;

            state.profile.weight = parseFloat(document.getElementById('weight').value) || 70;

            state.profile.height = parseFloat(document.getElementById('height').value) || 175;

            state.profile.activity = parseFloat(document.getElementById('activity').value) || 1.55;

            state.profile.goal = document.getElementById('goal').value;


            // BMR Calculation (Mifflin-St Jeor Equation)

            let bmr = (10 * state.profile.weight) + (6.25 * state.profile.height) - (5 * state.profile.age);

            if (state.profile.gender === 'male') {

                bmr += 5;

            } else {

                bmr -= 161;

            }

            state.profile.bmr = Math.round(bmr);


            // TDEE

            let tdee = bmr * state.profile.activity;

            state.profile.tdee = Math.round(tdee);


            // Goal Adjustment

            let targetKcal = tdee;

            let goalText = "Gewicht halten";


            switch (state.profile.goal) {

                case 'lose_fast':

                    targetKcal = tdee * 0.80;

                    goalText = "Starkes Kaloriendefizit (-20%)";

                    break;

                case 'lose':

                    targetKcal = tdee * 0.85;

                    goalText = "Moderates Kaloriendefizit (-15%)";

                    break;

                case 'maintain':

                    targetKcal = tdee;

                    goalText = "Erhaltungskalorien";

                    break;

                case 'gain':

                    targetKcal = tdee * 1.10;

                    goalText = "Moderater Überschuss (+10%)";

                    break;

                case 'gain_fast':

                    targetKcal = tdee * 1.15;

                    goalText = "Höherer Überschuss (+15%)";

                    break;

            }


            state.profile.targetKcal = Math.round(targetKcal);


            // Grams calculations based on % ratios

            // Protein: 4 kcal/g, Carbs: 4 kcal/g, Fat: 9 kcal/g

            const pKcal = state.profile.targetKcal * (state.profile.pPct / 100);

            const cKcal = state.profile.targetKcal * (state.profile.cPct / 100);

            const fKcal = state.profile.targetKcal * (state.profile.fPct / 100);


            state.profile.targetP = Math.round(pKcal / 4);

            state.profile.targetC = Math.round(cKcal / 4);

            state.profile.targetF = Math.round(fKcal / 9);


            saveStateToLocalStorage();

            recalculateProfileAndUI();

            

            // Switch back to dashboard on calculation save

            switchTab('dashboard');

        }

            Object.keys(state.log).forEach(mealKey => {

                let mealKcal = 0;

                state.log[mealKey].forEach(item => {

                    const food = state.foods.find(f => f.id === item.foodId);

                    if (food) {

                        const factor = item.amount / 100;

                        const kcal = food.kcal * factor;

                        const p = food.p * factor;

                        const c = food.c * factor;

                        const f = food.f * factor;


                        mealKcal += kcal;

                        totalKcal += kcal;

                        totalP += p;

                        totalC += c;

                        totalF += f;

                    }

                });

                const summaryEl = document.getElementById(`meal-summary-${mealKey}`);

                if (summaryEl) summaryEl.innerText = `${Math.round(mealKcal)} kcal`;

            });


            totalKcal = Math.round(totalKcal);

            totalP = Math.round(totalP);

            totalC = Math.round(totalC);

            totalF = Math.round(totalF);


            // Dashboard Numbers Update

            document.getElementById('consumed-kcal-display').innerText = `${totalKcal} kcal`;

            

            const remainingKcal = (state.profile.targetKcal || 0) - totalKcal;

            document.getElementById('kcal-remaining').innerText = remainingKcal;


            // Status Badge

            const statusBadge = document.getElementById('calorie-status-badge');

            if (remainingKcal >= 0) {

                statusBadge.innerText = 'Im Zielbereich';

                statusBadge.className = 'text-xs font-semibold px-2 py-0.5 rounded-full bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300';

            } else {

                statusBadge.innerText = 'Kalorienüberschritt!';

                statusBadge.className = 'text-xs font-semibold px-2 py-0.5 rounded-full bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300';

            }


            // Macros UI

            document.getElementById('p-eaten').innerText = totalP;

            document.getElementById('c-eaten').innerText = totalC;

            document.getElementById('f-eaten').innerText = totalF;


            // Bar Widths

            const pPct = state.profile.targetP ? Math.min(100, Math.round((totalP / state.profile.targetP) * 100)) : 0;

            const cPct = state.profile.targetC ? Math.min(100, Math.round((totalC / state.profile.targetC) * 100)) : 0;

            const fPct = state.profile.targetF ? Math.min(100, Math.round((totalF / state.profile.targetF) * 100)) : 0;


            document.getElementById('p-bar').style.width = `${pPct}%`;

            document.getElementById('c-bar').style.width = `${cPct}%`;

            document.getElementById('f-bar').style.width = `${fPct}%`;


            // Ratio distribution text

            const macroKcalSum = (totalP * 4) + (totalC * 4) + (totalF * 9);

            if (macroKcalSum > 0) {

                const ratioP = Math.round(((totalP * 4) / macroKcalSum) * 100);

                const ratioC = Math.round(((totalC * 4) / macroKcalSum) * 100);

                const ratioF = Math.round(((totalF * 9) / macroKcalSum) * 100);

                document.getElementById('macro-ratio-text').innerText = `${ratioP}% / ${ratioC}% / ${ratioF}%`;

            } else {

                document.getElementById('macro-ratio-text').innerText = `0% / 0% / 0%`;

            }


            updateCharts(totalKcal);

        }
